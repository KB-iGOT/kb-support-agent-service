# Cron Job Configuration for Enrollment Cleanup
# Add this to your crontab on the PostgreSQL VM

# Run enrollment cleanup every day at 2:00 AM
# This will delete enrollment records older than 2 days
0 2 * * * /path/to/scripts/cleanup_enrollments.sh

# Alternative schedules (uncomment the one you prefer):

# Run every 12 hours (2 AM and 2 PM)
# 0 2,14 * * * /path/to/scripts/cleanup_enrollments.sh

# Run every 6 hours
# 0 */6 * * * /path/to/scripts/cleanup_enrollments.sh

# Run weekly on Sunday at 3 AM
# 0 3 * * 0 /path/to/scripts/cleanup_enrollments.sh
